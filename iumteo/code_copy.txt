/**
 * ==========================================
 * 양양이음터 강사 및 이용자 DB 통합 연동 API (Next.js 호환)
 * 통합 기능: 강사 목록 조회, 로그인, 회원가입(강사/이용자), 프로필 수정 및 상태 변경
 * 
 * 배포 방법:
 * 1. 이 코드를 복사해서 구글 앱스 스크립트에 모두 붙여넣기
 * 2. 상단 '배포' -> '배포 관리' -> 연필 아이콘 -> 버전 '새 버전' 선택 후 저장
 * ==========================================
 */


const API_KEY = 'yy-iumteo-secret-key-2026'; 

/**
 * ------------------------------------------
 * 공통 유틸리티 함수
 * ------------------------------------------
 */
function normalizePhone(phone) {
  if (!phone) return '';
  return String(phone).replace(/-/g, '').trim();
}

function normalizeEmail(email) {
  if (!email) return '';
  return String(email).toLowerCase().replace(/\s+/g, '');
}

/**
 * 보안 검증 유틸리티 
 * 기존 회원가입 폼(프론트엔드 직접 전송)을 위해 apiKey가 없어도 통과하도록 하되, 
 * 민감한 조회(getUser)나 회원 정보 수정(updateInstructor)은 API Key를 강제합니다.
 */
function validateApiKey(requestKey, strictMode = false) {
  if (strictMode) {
    return requestKey === API_KEY;
  }
  // strictMode가 아니면 key가 없거나 일치할 때 통과 (기존 레거시 코드 호환용)
  return !requestKey || requestKey === API_KEY;
}

/**
 * 헤더 맵핑 유틸리티
 */
function getHeaderMap(sheet) {
  const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  const headerMap = {};
  headers.forEach((header, index) => {
    if (header) headerMap[header.trim()] = index;
  });
  return headerMap;
}

function rowToObject(row, headerMap) {
  const obj = {};
  for (const key in headerMap) {
    obj[key] = row[headerMap[key]];
  }
  return obj;
}

function getValueByAliases(row, headerMap, aliases) {
  for (let i = 0; i < aliases.length; i++) {
    const header = aliases[i];
    if (headerMap[header] !== undefined) {
      return row[headerMap[header]];
    }
  }
  return '';
}

function normalizeYesNoFlag(value) {
  if (value === true || value === 1) return 'Y';
  if (typeof value === 'string') {
    const normalized = value.trim().toUpperCase();
    if (['Y', 'YES', 'TRUE', '1'].includes(normalized)) return 'Y';
  }
  return 'N';
}

function normalizeBooleanFlag(value, fallback) {
  if (typeof value === 'boolean') return value;
  if (typeof value === 'number') return value === 1;
  if (typeof value === 'string') {
    const normalized = value.trim().toLowerCase();
    if (['true', 'y', 'yes', '1'].includes(normalized)) return true;
    if (['false', 'n', 'no', '0', ''].includes(normalized)) return false;
  }
  return fallback;
}

function normalizeStatusValue(value) {
  return typeof value === 'string' && value.trim() ? value.trim() : '\uB300\uAE30';
}

function normalizeInstructorRow(row, headerMap) {
  const obj = rowToObject(row, headerMap);
  const status = normalizeStatusValue(getValueByAliases(row, headerMap, ['status', '\uC0C1\uD0DC']));
  const isLocal = normalizeYesNoFlag(getValueByAliases(row, headerMap, ['isLocal', '\uB85C\uCEEC', '\uB85C\uCEEC\uC5EC\uBD80']));
  const isSample = normalizeBooleanFlag(getValueByAliases(row, headerMap, ['isSample']), true);

  obj.status = status;
  obj.isLocal = isLocal;
  obj.isSample = isSample;

  if (obj['\uC0C1\uD0DC'] === undefined || obj['\uC0C1\uD0DC'] === '') obj['\uC0C1\uD0DC'] = status;
  if (obj['\uB85C\uCEEC'] === undefined || obj['\uB85C\uCEEC'] === '') obj['\uB85C\uCEEC'] = isLocal;

  return obj;
}

function successResponse(msg, data = null) {
  const responseData = { success: true, status: "success", message: msg };
  if (data) responseData.data = data;
  return ContentService.createTextOutput(JSON.stringify(responseData))
    .setMimeType(ContentService.MimeType.JSON);
}

function errorResponse(msg, code = 400) {
  return ContentService.createTextOutput(JSON.stringify({ success: false, status: "error", message: msg, error: msg, code: code }))
    .setMimeType(ContentService.MimeType.JSON);
}


/**
 * ------------------------------------------
 * HTTP OPTIONS (CORS 방지용 프리플라이트)
 * ------------------------------------------
 */
function doOptions(e) {
  return ContentService.createTextOutput('')
    .setMimeType(ContentService.MimeType.TEXT);
}


/**
 * ------------------------------------------
 * GET 요청 핸들러 (데이터 조회: 강사 목록 가져오기, 로그인)
 * ------------------------------------------
 */
function doGet(e) {
  const action = e.parameter.action;
  const requestKey = e.parameter.apiKey;
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  try {
    // [액션 1 - GET] 강사 목록 조회 (프론트엔드 노출용)
    if (action === 'getInstructors') {
      const sheet = ss.getSheetByName('\uC591\uC591\uC774\uC74C\uD130\uAC15\uC0ACDB');
      if (!sheet) throw new Error("Sheet '\uC591\uC591\uC774\uC74C\uD130\uAC15\uC0ACDB' not found");
      const data = sheet.getDataRange().getValues();
      const headerMap = getHeaderMap(sheet);
      const includeAll = e.parameter.includeAll === 'Y' || e.parameter.includeAll === 'true';
      const canViewAll = includeAll && validateApiKey(requestKey, true);

      const instructors = [];
      for (let i = 1; i < data.length; i++) {
        const row = data[i];
        const status = normalizeStatusValue(getValueByAliases(row, headerMap, ['status', '\uC0C1\uD0DC']));
        const consent = getValueByAliases(row, headerMap, ['\uB3D9\uC758\uC5EC\uBD80', 'consent']);

        if (canViewAll || (status === '\uC2B9\uC778' && consent === 'Y')) {
          instructors.push(normalizeInstructorRow(row, headerMap));
        }
      }
      return successResponse("Instructors fetched", instructors);
    }
    
    // [액션 2 - GET] 특정 사용자/강사 정보 반환 (Next.js 로그인용)
    // 이 기능은 백엔드 통신 전용이므로 Strict API Key 검증
    if (action === 'getUser') {
      if (!validateApiKey(requestKey, true)) return errorResponse("Unauthorized API Key", 401);

      const targetEmail = normalizeEmail(e.parameter.email);
      if (!targetEmail) throw new Error('Email parameter required');

      // 1. 강사 DB부터 확인
      let sheet = ss.getSheetByName('양양이음터강사DB');
      if (sheet) {
        let data = sheet.getDataRange().getValues();
        let headerMap = getHeaderMap(sheet);
        
        for (let i = 1; i < data.length; i++) {
          const row = data[i];
          const currentEmail = normalizeEmail(row[headerMap['\uB85C\uADF8\uC778\uC6A9 \uC774\uBA54\uC77C']]);
          if (currentEmail === targetEmail) {
            const userObj = normalizeInstructorRow(row, headerMap);
            userObj._role = 'INSTRUCTOR'; 
            userObj._rowIndex = i + 1;
            return successResponse("Instructor found", userObj);
          }
        }
      }

      // 2. 관리자/일반 이용자 DB 확인 
      sheet = ss.getSheetByName('이용자DB'); 
      if(sheet) {
        let data = sheet.getDataRange().getValues();
        let headerMap = getHeaderMap(sheet);
        
        for (let i = 1; i < data.length; i++) {
          const row = data[i];
          const currentEmail = normalizeEmail(row[headerMap['이메일(로그인용)']]);
          if (currentEmail === targetEmail) {
            const userObj = rowToObject(row, headerMap);
            const roleHeaderIndex = headerMap['권한'] !== undefined ? headerMap['권한'] : headerMap['권한(USER)'];
            userObj._role = (roleHeaderIndex !== undefined ? row[roleHeaderIndex] : null) || 'USER';
            userObj._rowIndex = i + 1;
            return successResponse("User found", userObj);
          }
        }
      }

      // 3. 테스트용 어드민 하드코딩 패스 (Fallback)
      if (targetEmail === 'admin@yangyang.go.kr' || targetEmail === 'admin') {
         return successResponse("Test Admin Fallback", {
             "이메일(로그인용)": "admin",
             "이용자명": "어드민(테스트)",
             "비밀번호": "admin",
             "권한": "ADMIN",
             "_role": "ADMIN",
             "_rowIndex": 0
         });
      }

      return errorResponse("User not found", 404);
    }

    return errorResponse("Invalid GET action");

  } catch (err) {
    return errorResponse(err.message);
  }
}


/**
 * ------------------------------------------
 * POST 요청 핸들러 (데이터 업데이트/수정/생성)
 * 기존 회원가입 로직 통합 + 새로운 보안/맵핑 로직
 * ------------------------------------------
 */
function doPost(e) {
  try {
    let payload;
    try {
      payload = JSON.parse(e.postData.contents);
    } catch (parseErr) {
      throw new Error("JSON Parsing Failed: " + parseErr.message);
    }

    const action = payload.action;
    const requestKey = payload.apiKey;
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // ====================================================
    // [액션 3 - POST] 강사 신규 프로필 등록 (join)
    // 기존 호환: 엄격한 API Key 검증 제외 (클라이언트 직접 호출)
    // ====================================================
    if (action === "join") {
      const sheet = ss.getSheetByName("\uC591\uC591\uC774\uC74C\uD130\uAC15\uC0ACDB");
      if (!sheet) throw new Error("'\uC591\uC591\uC774\uC74C\uD130\uAC15\uC0ACDB' \uC2DC\uD2B8\uB97C \uCC3E\uC744 \uC218 \uC5C6\uC2B5\uB2C8\uB2E4.");

      const headerMap = getHeaderMap(sheet);
      const rowLength = Math.max(sheet.getLastColumn(), 18);
      const newRow = new Array(rowLength).fill("");

      function setCell(headerName, value) {
        if (headerMap[headerName] !== undefined) {
          newRow[headerMap[headerName]] = value;
        }
      }

      setCell('\uC131\uBA85', payload.name || '');
      setCell('\uC18C\uC18D', payload.org || '');
      setCell('\uC0C1\uC138\uB0B4\uC6A9', '');
      setCell('\uC8FC\uC694\uACBD\uB825', payload.career || '');
      setCell('\uC774\uBA54\uC77C', payload.email || '');
      setCell('\uC778\uC2A4\uD0C0\uADF8\uB7A8', payload.instagram || '');
      setCell('\uC778\uC2A4\uD0C0\uADF8\uB7A8\uACF5\uAC1C\uC5EC\uBD80', payload.insta_open || '');
      setCell('\uAC15\uC758\uBD84\uC57C', payload.field || '');
      setCell('\uB3D9\uC758\uC5EC\uBD80', payload.info_open_agree || '');
      setCell('\uB85C\uCEEC', '');
      setCell('\uC5F0\uB77D\uCC98', payload.phone || '');
      setCell('\uC8FC\uC18C', payload.address || '');
      setCell('\uC0AC\uC6A9\uC790\uBE44\uBC88', payload.password || '');
      setCell('\uD504\uB85C\uD544\uC0AC\uC9C4', payload.profile_photo || '');
      setCell('\uC0C1\uD0DC', payload.status || '\uB300\uAE30');
      setCell('\uAC15\uC0ACID', '');
      setCell('\uB85C\uADF8\uC778\uC6A9 \uC774\uBA54\uC77C', payload.email || '');
      setCell('isSample', false);

      if (headerMap['isSample'] === undefined && newRow.length >= 18) {
        newRow[17] = false;
      }

      sheet.appendRow(newRow);
      return successResponse("\uAC15\uC0AC \uB4F1\uB85D \uC644\uB8CC");
    }

    // ====================================================
    // [액션 4 - POST] 일반 이용자 회원 가입 (joinUser)
    // ====================================================
    if (action === "joinUser") {
      const sheet = ss.getSheetByName("이용자DB");
      if (!sheet) throw new Error("'이용자DB' 시트를 찾을 수 없습니다.");
      
      const dateStr = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy. M. d.");
      const newRow = [
        dateStr,                       // A 가입일
        payload.name || "",            // B 이용자명
        payload.email || "",           // C 이메일(로그인용)
        payload.org || "",             // D 소속명
        payload.phone || "",           // E 연락처
        payload.password || "",        // F 비밀번호
        "USER",                        // G 권한 (기본값)
        payload.memberType || "이용자", // H 회원유형
        "정상"                         // I 상태
      ];
      sheet.appendRow(newRow);
      return successResponse("이용자 가입 완료");
    }

    // ====================================================
    // [액션 5 - POST] 관리자: 상태 변경 (update_status)
    // 기존 호환, 하지만 프론트엔드 연동을 위해 헤더 맵핑 적용 추천
    // ====================================================
    if (action === "update_status") {
      // 본 기능은 관리자 기능이므로 향후 Strict API Key 적용 권장
      const sheet = ss.getSheetByName("양양이음터강사DB");
      const rowIndex = payload.rowIndex; 
      const status = payload.status;
      sheet.getRange(rowIndex, 15).setValue(status); // 'O'열 하드코딩 유지 (기존 프론트엔드 파괴 방지)
      return successResponse(`상태가 ${status}(으)로 변경되었습니다.`);
    }

    // ====================================================
    // [액션 6 - POST] 프로필 사진 변경 (update_photo)
    // ====================================================
    if (action === "update_photo") {
      const sheet = ss.getSheetByName("양양이음터강사DB");
      const rowIndex = payload.rowIndex;
      const photoUrl = payload.photoUrl;
      sheet.getRange(rowIndex, 14).setValue(photoUrl); // 'N'열 하드코딩 유지
      return successResponse("사진 업데이트 완료");
    }

    // ====================================================
    // [액션 7 - POST] Next.js 벡앤드 연동: 안전한 본인 프로필 수정 (updateInstructor)
    // - Strict API Key 탑재
    // - 인덱스 번호가 아닌 "이메일" 기반의 Row-level 보안 처리
    // - 헤더 이름을 기반으로 동적 맵핑 (기존 update_info 대체 모델)
    // ====================================================
    if (action === 'updateInstructor') {
      if (!validateApiKey(requestKey, true)) return errorResponse("Unauthorized API Key", 401);

      const targetEmail = normalizeEmail(payload.email);
      const updates = payload.updates; 
      
      const sheet = ss.getSheetByName('양양이음터강사DB');
      if (!sheet) throw new Error("Template mapping error: Sheet not found");
      
      const data = sheet.getDataRange().getValues();
      const headerMap = getHeaderMap(sheet);
      
      let targetRowIndex = -1;

      // Email 식별자로 본인 데이터 탐색
      for (let i = 1; i < data.length; i++) {
        if (normalizeEmail(data[i][headerMap['로그인용 이메일']]) === targetEmail) {
          targetRowIndex = i + 1; 
          break;
        }
      }

      if (targetRowIndex === -1) {
        return errorResponse("Row not found or Edit Unauthorized", 403);
      }

      // 업데이트 로직 적용 (헤더 이름 기반)
      for (const colName in updates) {
        if (headerMap.hasOwnProperty(colName)) {
          let newValue = updates[colName];
          if (colName === '연락처') newValue = normalizePhone(newValue);
          if (colName === '이메일' || colName === '로그인용 이메일') newValue = normalizeEmail(newValue);
          
          const colIndex = headerMap[colName] + 1;
          sheet.getRange(targetRowIndex, colIndex).setValue(newValue);
        }
      }

      return successResponse("Updated successfully");
    }

    // ====================================================
    // [액션 8 - POST] 관리자 지정 직접 수정 (기존 update_info 로직 하위 호환성)
    // ====================================================
    if (action === "update_info") {
      const sheet = ss.getSheetByName("양양이음터강사DB");
      const rowIndex = payload.rowIndex;
      
      if (payload.name !== undefined) sheet.getRange(rowIndex, 1).setValue(payload.name);
      if (payload.org !== undefined) sheet.getRange(rowIndex, 2).setValue(payload.org);
      if (payload.career !== undefined) sheet.getRange(rowIndex, 4).setValue(payload.career);
      if (payload.email !== undefined) sheet.getRange(rowIndex, 5).setValue(payload.email);
      if (payload.instagram !== undefined) sheet.getRange(rowIndex, 6).setValue(payload.instagram);
      if (payload.insta_open !== undefined) sheet.getRange(rowIndex, 7).setValue(payload.insta_open);
      if (payload.field !== undefined) sheet.getRange(rowIndex, 8).setValue(payload.field);
      if (payload.info_open_agree !== undefined) sheet.getRange(rowIndex, 9).setValue(payload.info_open_agree);
      if (payload.phone !== undefined) sheet.getRange(rowIndex, 11).setValue(payload.phone);
      if (payload.address !== undefined) sheet.getRange(rowIndex, 12).setValue(payload.address);
      if (payload.password !== undefined) sheet.getRange(rowIndex, 13).setValue(payload.password);
      
      return successResponse("강사 정보 업데이트 완료 (Legacy)");
    }

    return errorResponse("Invalid POST action");

  } catch (error) {
    return errorResponse(error.message);
  }
}
