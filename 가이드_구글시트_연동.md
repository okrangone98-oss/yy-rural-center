# 홈페이지 데이터 관리 시트 연동 가이드

## 1. Apps Script 설정 및 코드 적용

1. **[홈페이지 데이터 관리 시트]**(<https://docs.google.com/spreadsheets/d/1PMi4ajeKeY_aY5Sh9zBy4kWw8tz7pwQoH9uOLa1A6ag/edit)를> 엽니다.
2. 상단 메뉴에서 **[확장 프로그램] -> [Apps Script]**를 클릭합니다.
3. 열린 편집기의 기존 코드를 모두 지우고, 아래의 **전체 코드를 복사하여 붙여넣습니다**.

```javascript
const SPREADSHEET_ID = '1PMi4ajeKeY_aY5Sh9zBy4kWw8tz7pwQoH9uOLa1A6ag';
const SHEET_NAME = '강사관리'; // 실제 시트 하단 탭 이름과 일치해야 합니다. (예: '강사관리' 또는 '양양이음터강사DB')

function doGet(e) {
  const sheetName = e.parameter.sheet || SHEET_NAME;
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName(sheetName);
    if (!sheet) return createJsonResponse({ error: `시트 '${sheetName}'를 찾을 수 없습니다.` });

    const data = sheet.getDataRange().getValues();
    const headers = data[0];
    const rows = data.slice(1);
    
    // 기본적으로 최신순으로 반환
    const result = rows.reverse().map((row, idx) => {
      let obj = { rowIndex: rows.length - idx + 1 };
      headers.forEach((header, index) => {
        let val = row[index];
        if (val instanceof Date) {
          val = Utilities.formatDate(val, "GMT+9", "yyyy-MM-dd");
        }
        obj[header] = val;
      });
      return obj;
    });
    
    return createJsonResponse({ success: true, data: result });
  } catch (error) {
    return createJsonResponse({ error: error.message });
  }
}

function doPost(e) {
  try {
    let requestData;
    if (e.postData && e.postData.contents) {
      requestData = JSON.parse(e.postData.contents);
    } else {
      requestData = e.parameter;
    }

    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sheet = ss.getSheetByName(SHEET_NAME);
    if (!sheet) return createJsonResponse({ success: false, error: "시트를 찾을 수 없습니다." });
    
    const headers = sheet.getDataRange().getValues()[0];

    // --- 1. 삭제 (Delete) ---
    if (requestData.action === 'delete' && requestData.rowIndex) {
      const rowIndex = parseInt(requestData.rowIndex);
      sheet.deleteRow(rowIndex);
      return createJsonResponse({ success: true, message: "삭제 완료" });
    }

    // --- 2. 수정 (Update / Approve) ---
    if (requestData.action === 'update' && requestData.rowIndex) {
      const rowIndex = parseInt(requestData.rowIndex);
      // 전달된 모든 필드를 헤더 기준으로 업데이트
      for (let key in requestData) {
        if (['action', 'rowIndex'].includes(key)) continue;
        
        // 영문 필드명을 한글 헤더로 매핑 (필요시)
        const headerName = getHeaderName(key);
        const colIndex = headers.indexOf(headerName) + 1;
        
        if (colIndex > 0) {
          sheet.getRange(rowIndex, colIndex).setValue(requestData[key]);
        }
      }
      return createJsonResponse({ success: true, message: "수정 완료" });
    }

    // --- 3. 신규 신청 (Insert) ---
    const newRow = headers.map(header => {
      // 타임스탬프
      if (['타임스탬프', '작성일', '제출일'].includes(header)) return new Date();
      
      // 필드 매핑 (영문 -> 한글)
      if (header === '성명') return requestData.name || '';
      if (header === '소속') return requestData.org || '';
      if (header === '상세내용') return requestData.field_detail || '';
      if (header === '주요경력') return requestData.career || '';
      if (header === '이메일') return requestData.email || '';
      if (header === '인스타그램주소') return requestData.instagram || '';
      if (header === '인스타그램공개여부') return requestData.insta_open || '';
      if (header === '강의분야') return requestData.field || ''; // 폼의 field가 시트의 강의분야
      if (header === '동의여부') return requestData.info_open_agree || '';
      if (header === '연락처') return requestData.phone || '';
      if (header === '주소') return requestData.address || '';
      if (header === '사용자비번') return requestData.password || '';
      if (header === '프로필사진') return requestData.profile_photo || '';
      if (header === '상태') return requestData.status || '대기';
      if (header === '로컬') return requestData.isLocal || 'N';
      
      return requestData[header] || '';
    });

    sheet.appendRow(newRow);
    return createJsonResponse({ success: true, message: "등록 완료" });

  } catch (error) {
    return createJsonResponse({ success: false, error: error.message });
  }
}

// 영문 필드명을 시트의 한글 헤더명으로 변환하는 도우미 함수
function getHeaderName(key) {
  const map = {
    'name': '성명', 'org': '소속', 'phone': '연락처', 'email': '이메일',
    'field': '강의분야', 'field_detail': '상세내용', 'career': '주요경력',
    'instagram': '인스타그램주소', 'insta_open': '인스타그램공개여부',
    'address': '주소', 'password': '사용자비번', 'profile_photo': '프로필사진',
    'info_open_agree': '동의여부', 'isLocal': '로컬', 'status': '상태'
  };
  return map[key] || key;
}

function createJsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
```

## 2. 웹 앱(Web App) 배포하기

1. 코드 편집기 상단의 디스크 아이콘(저장)을 눌러 저장합니다.
2. 우측 상단의 **[배포] -> [새 배포]**를 클릭합니다.
3. 톱니바퀴 아이콘을 누르고 **웹 앱**을 선택합니다.
4. 설명에 "홈페이지 연동 API"라고 적습니다.
5. `다음 사용자로서 실행(Execute As)`: **나(웹 앱 소유자)**
6. `액세스할 수 있는 사용자(Who has access)`: **모든 사용자(Anyone)**
7. **[배포]** 버튼을 클릭합니다.
8. (최초 배포 시) '액세스 승인' 창이 나오면 권한을 허용해줍니다.
   - 계정 선택 -> 고급 -> *(안전하지 않음)으로 이동 -> Allow(허용)
9. 생성된 **웹 앱 URL(Web App URL)**을 복사합니다.

## 3. 프론트엔드 연동 대기

- 발급받으신 **웹 앱 URL**을 복사해서 대화창에 남겨주시면, 홈페이지의 `index.html` 내에 연동하는 작업을 이어가겠습니다!
